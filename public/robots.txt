# The County Post — https://thecountypost.com
# Local news, weather, and official county data for all 3,143 U.S. counties.

# ---------------------------------------------------------------------------
# Default policy: closed. Only the search engines named below may crawl.
#
# AI training and answer-engine crawlers, SEO-tool crawlers and scrapers fall
# under this group. Crawlers that render the site call the news API for every
# county they visit, and most of those counties are not pre-built, so each
# visit is a slow, uncached build paid for by The County Post.
# ---------------------------------------------------------------------------
User-agent: *
Disallow: /

# ---------------------------------------------------------------------------
# AI-training opt-outs for the search engines allowed below. These tokens are
# never crawlers themselves: they control whether Google and Apple may use
# what Googlebot and Applebot fetch to train AI models. Search is unaffected.
# ---------------------------------------------------------------------------
User-agent: Google-Extended
User-agent: Applebot-Extended
Disallow: /

# ---------------------------------------------------------------------------
# Search engines.
#
# Googlebot also covers Googlebot-Image, Googlebot-News and Googlebot-Video,
# which fall back to the Googlebot group. Google-InspectionTool runs Search
# Console's URL Inspection and Rich Results tests.
# ---------------------------------------------------------------------------
User-agent: Googlebot
User-agent: Google-InspectionTool
User-agent: Bingbot
User-agent: DuckDuckBot
User-agent: Applebot
Allow: /

# Submission and classified forms carry no standalone content and exist only to
# post to EmailJS. They are also marked noindex in the page itself.
Disallow: /submit
Disallow: /*/submit
Disallow: /*/*/submit
Disallow: /*/*/classifieds

# Never useful to a crawler.
Disallow: /*?checkout=
Disallow: /*?*utm_

# ---------------------------------------------------------------------------
# Not crawlers: services that fetch a single page on request. Link-preview
# fetchers build the card shown when a story is shared, and Stripebot checks
# the site for the Stripe account behind advertiser checkout.
# ---------------------------------------------------------------------------
User-agent: Twitterbot
User-agent: facebookexternalhit
User-agent: LinkedInBot
User-agent: Slackbot
User-agent: Discordbot
User-agent: Stripebot
Allow: /
